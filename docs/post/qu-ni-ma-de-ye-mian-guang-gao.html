去你妈的页面广告

用于解决一些视频网站的广告问题

// ==UserScript==
// @name         视频网站广告拦截终极版
// @namespace    http://tampermonkey.net/
// @version      6.1
// @description  拦截视频网站底部横幅广告、顶部广告、隐形遮罩广告、动态JS广告注入框架、Yandex统计追踪器，同时保护视频播放器与顶部导航菜单（主页/搜索/动漫等）不被误杀；支持网站白名单控制
// @author       You
// @match        *://*/*
// @icon         data:image/gif;base64,R0lGODlhAQABAAAAACH5BAEKAAEALAAAAAABAAEAAAICTAEAOw==
// @grant        none
// @run-at       document-start
// ==/UserScript==

(function () {
    'use strict';

    // ===================== 6.0 新增：网站白名单 =====================
    // 只有域名匹配白名单中的站点才会运行脚本，防止误伤其他网站
    const WEBSITE_WHITELIST = [
        '修改这个位置，只保留主域名即可',       //不用带https或者http，直接把域名填进去就行了
        // 在此添加更多允许运行的网站域名，例如：
        // 'example.com',
        // 'yourvideosite.com',
    ];

    // 域名匹配检查
    const currentHost = window.location.hostname;
    const isAllowed = WEBSITE_WHITELIST.some(domain =>
        currentHost === domain || currentHost.endsWith('.' + domain)
    );
    if (!isAllowed) {
        // 不在白名单中，直接退出，不执行任何操作
        return;
    }

    // ===================== 配置项 =====================
    const CONFIG = {
        DEBUG: false,
        SCAN_DELAY: [500, 1000, 2000, 5000],
        DEBOUNCE_MS: 150
    };

    // ===================== 日志工具 =====================
    const log = (...args) => CONFIG.DEBUG && console.log('[AdBlock-v6]', ...args);
    const warn = (...args) => CONFIG.DEBUG && console.warn('[AdBlock-v6]', ...args);

    // ===================== 恶意特征 列表 =====================
    const MALICIOUS_SCRIPT_PATTERNS = [
        'idia', 'mc.yandex', 'metrika', 'tag.js', 'watch/',
        'bcvxfsd68', '1-727-xz-xuqnmqvgnxa', '94684572994-teuhjsexgjj',
        'wwnb8', '2dnmKvC64iojTSyNURSwuD', '2876'
    ];

    const MALICIOUS_FETCH_DOMAINS = [
        'mc.yandex.ru', 'bcvxfsd68', '1-727-xz-xuqnmqvgnxa',
        '94684572994-teuhjsexgjj', 'wwnb8', '2dnmKvC64iojTSyNURSwuD'
    ];

    const AD_TEXT_PATTERNS = [
        '经常点到广告', '收藏下面任意发布页地址', '花都发布页', 'b.hdflby'
    ];

    // ===================== 站点导航/顶栏保护（防顶部菜单误杀） =====================
    // 顶部"个人主页 / 搜索 / 主页 / 动漫 / 国产"等导航容器的保护规则
    // 命中任一规则的固定/绝对定位顶栏将被整体跳过，不再当广告移除
    const NAV_PROTECT_SELECTORS = [
        // 通用标签与语义化容器
        'nav', 'header', '[role="navigation"]', '[role="banner"]',
        // 常见顶栏 class 关键词（按需补充你自己站点的选择器）
        '[class*="navbar"]', '[class*="nav-bar"]', '[class*="topbar"]',
        '[class*="top-bar"]', '[class*="header"]', '[class*="site-header"]',
        '[class*="site-nav"]', '[class*="main-nav"]', '[class*="primary-nav"]',
        '[class*="menu"]', '[class*="topnav"]', '[class*="head-nav"]',
        '[class*="stui-header"]', '[class*="vod-header"]', '[class*="w-header"]',
        // 常见顶栏 id
        '#navbar', '#header', '#site-header', '#main-nav', '#topbar', '#topBar'
    ];
    const NAV_PROTECT_SELECTOR_STR = NAV_PROTECT_SELECTORS.join(', ');

    // 判断元素是否属于站点顶部导航/菜单区（整体保护，不误杀）
    function isNavigationElement(el) {
        if (!el || el.nodeType !== 1) return false;
        // 1) 直接命中导航选择器
        try {
            if (el.matches && el.matches(NAV_PROTECT_SELECTOR_STR)) return true;
        } catch (e) { }
        // 2) 自身位于导航容器内（向上查找）
        try {
            if (el.closest && el.closest(NAV_PROTECT_SELECTOR_STR)) return true;
        } catch (e) { }
        // 3) 固定/绝对定位且紧贴顶部，内部含菜单链接/logo/搜索框 → 视为导航
        try {
            const style = window.getComputedStyle(el);
            if ((style.position === 'fixed' || style.position === 'absolute') &&
                parseFloat(style.top) <= 10) {
                if (el.querySelector &&
                    (el.querySelector('a[href]') || el.querySelector('input[type="search"]') ||
                     el.querySelector('.logo, [class*="logo"], [class*="brand"], [class*="search"]'))) {
                    return true;
                }
            }
        } catch (e) { }
        return false;
    }

    // ===================== 视频播放器保护 =====================
    const PLAYER_SELECTORS = [
        'video', 'iframe[src*=\"player\"]', 'iframe[src*=\"embed\"]',
        '.video-container', '.player-container', '#player',
        '.dplayer', '.xgplayer', '.art-video-player',
        '.prism-player', '.txp_video_container', '.aliplayer',
        '.bpx-player-container', '.bilibili-player',
        '.youku-player', '.iqiyi-player', '.qqplayer',
        '[class*=\"player-wrap\"]', '[class*=\"video-wrap\"]',
        '[id*=\"player\"]', '[id*=\"video-player\"]'
    ];

    const PLAYER_SELECTOR_STR = PLAYER_SELECTORS.join(', ');

    function isPlayerElement(el) {
        if (!el || el.nodeType !== 1) return false;
        try {
            if (el.matches && el.matches(PLAYER_SELECTOR_STR)) return true;
        } catch (e) { }
        if (el.querySelector && el.querySelector(PLAYER_SELECTOR_STR)) return true;
        let parent = el.parentElement;
        let depth = 0;
        while (parent && depth < 8) {
            try {
                if (parent.matches && parent.matches(PLAYER_SELECTOR_STR)) return true;
            } catch (e) { }
                if (parent.querySelector && parent.querySelector(PLAYER_SELECTOR_STR)) return true;
            parent = parent.parentElement;
            depth++;
        }
        return false;
    }

    // ===================== 6.0 新增：视频预览图白名单检测 =====================
    // 防止视频列表中的缩略图被误删
    function isVideoThumbnail(el) {
        if (!el || el.nodeType !== 1) return false;

        // 判断是否是 img.lazyload 且位于 stui-vodlist 等视频列表结构中
        if (el.tagName === 'IMG' && el.classList && el.classList.contains('lazyload')) {
            const parent = el.closest('ul.stui-vodlist, ul.stui-vodlist__detail, .vodlist, .stui-vodlist');
            if (parent) return true;
        }

        // 判断是否在 stui-vodlist__thumb / w-thumb / picture 等缩略图容器中
        const thumb = el.closest('.stui-vodlist__thumb, .w-thumb, .picture');
        if (thumb && el.tagName === 'IMG') return true;

        // 判断是否在视频列表项 li 中（如 li.col-md-5.col-sm-4.col-xs-2）
        const videoListItem = el.closest('li.stui-vodlist__item, li.col-md-5, li.col-sm-4, li.col-xs-2');
        if (videoListItem && el.tagName === 'IMG') return true;

        // 判断是否在 stui-vodlist 包裹的 a 标签链中（a.stui-vodlist__thumb.picture.w-thumb）
        const thumbLink = el.closest('a.stui-vodlist__thumb, a.picture, a.w-thumb');
        if (thumbLink && el.tagName === 'IMG') return true;

        return false;
    }

    function safeRemove(el, reason) {
        if (!el || !el.tagName) return false;
        if (isPlayerElement(el)) {
            log('跳过受保护元素 [' + reason + ']:', el.tagName, el.className || el.id);
            return false;
        }
        // 6.0 新增：跳过视频预览图
        if (isVideoThumbnail(el)) {
            log('跳过视频预览图 [' + reason + ']:', el.tagName, el.className || el.id);
            return false;
        }
        // 6.1 新增：跳过站点顶部导航/菜单容器（防误杀 主页/搜索/动漫 等）
        if (isNavigationElement(el)) {
            log('跳过导航元素 [' + reason + ']:', el.tagName, el.className || el.id);
            return false;
        }
        log('移除元素 [' + reason + ']:', el.tagName, el.className || el.id || el.src || '');
        el.remove();
        return true;
    }

    function isAdScriptSrc(src) {
        if (!src) return false;
        const lower = src.toLowerCase();
        return MALICIOUS_SCRIPT_PATTERNS.some(p => lower.includes(p));
    }

    function isAdFetchUrl(url) {
        if (!url) return false;
        const lower = url.toLowerCase();
        return MALICIOUS_FETCH_DOMAINS.some(d => lower.includes(d));
    }

    function containsAdText(el) {
        if (!el || !el.textContent) return false;
        const text = el.textContent;
        return AD_TEXT_PATTERNS.some(p => text.includes(p));
    }

    // ===================== 1. 锁定 window.idia =====================
    try {
        Object.defineProperty(window, 'idia', {
            configurable: false,
            enumerable: false,
            get: function () { return null; },
            set: function () { log('Blocked window.idia assignment'); }
        });
    } catch (e) { warn('Failed to lock window.idia:', e); }

    // ===================== 2. 删除 Yandex Metrika =====================
    try {
        delete window.ym;
        Object.defineProperty(window, 'ym', {
            configurable: false,
            enumerable: false,
            get: function () { return function () {}; },
            set: function () { log('Blocked window.ym assignment'); }
        });
    } catch (e) { warn('Failed to lock window.ym:', e); }

    // ===================== 3. 劫持 document.createElement =====================
    const _origCreateElement = document.createElement.bind(document);
    document.createElement = function (tagName, options) {
        if (tagName && tagName.toLowerCase() === 'script') {
            const fake = _origCreateElement.call(document, 'div', options);
            let _src = '';
            Object.defineProperty(fake, 'src', {
                get: () => _src,
                set: (val) => {
                    _src = val;
                    if (isAdScriptSrc(val)) {
                        log('Blocked script src:', val);
                        return;
                    }
                    const realScript = _origCreateElement.call(document, 'script', options);
                    realScript.src = val;
                    if (fake.parentNode) {
                        fake.parentNode.replaceChild(realScript, fake);
                    }
                },
                configurable: true
            });
            return fake;
        }
        return _origCreateElement.call(document, tagName, options);
    };
    // ===================== 4. 劫持 document.write / writeln =====================
    const _origWrite = document.write.bind(document);
    const _origWriteln = document.writeln.bind(document);

    function sanitizeWrite(fn, args) {
        fn.apply(document, args);
        requestAnimationFrame(() => scanAndRemoveAds());
    }

    document.write = function () {
        const joined = Array.from(arguments).join('');
        if (joined.includes('idia') || joined.includes('mc.yandex') || joined.includes('metrika')) {
            log('Blocked document.write with ad content');
            return;
        }
        sanitizeWrite(_origWrite, Array.from(arguments));
    };

    document.writeln = function () {
        const joined = Array.from(arguments).join('');
        if (joined.includes('idia') || joined.includes('mc.yandex') || joined.includes('metrika')) {
            log('Blocked document.writeln with ad content');
            return;
        }
        sanitizeWrite(_origWriteln, Array.from(arguments));
    };

    // ===================== 5. 劫持 eval =====================
    const _origEval = window.eval;
    window.eval = function (code) {
        if (typeof code === 'string') {
            if (code.includes('idia') || code.includes('mc.yandex') ||
                code.includes('2dnmKvC64iojTSyNURSwuD') || code.includes('wwnb8')) {
                log('Blocked eval with ad code');
                return;
            }
        }
        return _origEval.call(window, code);
    };

    // ===================== 6. 劫持 fetch =====================
    const _origFetch = window.fetch;
    window.fetch = function (input, init) {
        const url = typeof input === 'string' ? input : (input && input.url ? input.url : '');
        if (isAdFetchUrl(url)) {
            log('Blocked fetch to:', url);
            return Promise.reject(new Error('Blocked by AdBlock-v6'));
        }
        return _origFetch.apply(this, arguments);
    };

    // ===================== 7. 劫持 XMLHttpRequest =====================
    const _origXHROpen = XMLHttpRequest.prototype.open;
    XMLHttpRequest.prototype.open = function (method, url) {
        if (isAdFetchUrl(url)) {
            log('Blocked XHR to:', url);
            this._adBlockBlocked = true;
        }
        return _origXHROpen.apply(this, arguments);
    };
    const _origXHRSend = XMLHttpRequest.prototype.send;
    XMLHttpRequest.prototype.send = function () {
        if (this._adBlockBlocked) {
            log('Blocked XHR send');
            return;
        }
        return _origXHRSend.apply(this, arguments);
    };

    // ===================== 8. 劫持 window.open =====================
    const _origWindowOpen = window.open;
    window.open = function (url) {
        if (url && isAdScriptSrc(url)) {
            log('Blocked window.open:', url);
            return null;
        }
        return _origWindowOpen.apply(this, arguments);
    };

    // ===================== 9. 劫持 addEventListener =====================
    const _origAddEventListener = EventTarget.prototype.addEventListener;
    EventTarget.prototype.addEventListener = function (type, listener, options) {
        if ((type === 'click' || type === 'touchstart' || type === 'touchend' ||
             type === 'mousedown' || type === 'mouseup') &&
            typeof listener === 'function') {
            const fnStr = listener.toString();
            if (fnStr.includes('location') || fnStr.includes('window.open') ||
                fnStr.includes('href') || fnStr.includes('navigate')) {
                if (!isPlayerElement(this)) {
                    log('Blocked suspicious event listener on', type, 'for', this);
                    return;
                }
            }
        }
        return _origAddEventListener.apply(this, arguments);
    };

    // ===================== 10. 劫持 innerHTML setter =====================
    const _htmlDescriptor = Object.getOwnPropertyDescriptor(Element.prototype, 'innerHTML');
    if (_htmlDescriptor && _htmlDescriptor.set) {
        Object.defineProperty(Element.prototype, 'innerHTML', {
            get: _htmlDescriptor.get,
            set: function (value) {
                const result = _htmlDescriptor.set.call(this, value);
                if (typeof value === 'string' &&
                    (value.includes('idia') || value.includes('banner-src') ||
                     value.includes('sponsor-block') || value.includes('mc.yandex'))) {
                    requestAnimationFrame(() => scanAndRemoveAds());
                }
                return result;
            },
            configurable: true,
            enumerable: true
        });
    }

    // ===================== 11. 劫持 appendChild / insertAdjacentElement =====================
    const _origAppendChild = Node.prototype.appendChild;
    Node.prototype.appendChild = function (child) {
        const result = _origAppendChild.apply(this, arguments);
        if (child && child.nodeType === 1) {
            debouncedScan();
        }
        return result;
    };

    const _origInsertAdjacentElement = Element.prototype.insertAdjacentElement;
    if (_origInsertAdjacentElement) {
        Element.prototype.insertAdjacentElement = function (position, element) {
            const result = _origInsertAdjacentElement.apply(this, arguments);
            if (element && element.nodeType === 1) {
                debouncedScan();
            }
            return result;
        };
    }

    // ===================== 12. 劫持 MutationObserver =====================
    const _OrigMutationObserver = window.MutationObserver;
    const _origObserve = _OrigMutationObserver.prototype.observe;

    _OrigMutationObserver.prototype.observe = function (target, options) {
        const result = _origObserve.apply(this, arguments);
        if (!this._adBlockInternalObserver) {
            this._adBlockInternalObserver = new _OrigMutationObserver((mutations) => {
                debouncedScan();
            });
            this._adBlockInternalObserver.observe(target, {
                childList: true,
                subtree: true,
                attributes: true,
                attributeFilter: ['class', 'style', 'src']
            });
        }
        return result;
    };

    // ===================== 广告扫描与移除核心逻辑 =====================
    let scanTimer = null;

    function debouncedScan() {
        if (scanTimer) return;
        scanTimer = setTimeout(() => {
            scanTimer = null;
            scanAndRemoveAds();
        }, CONFIG.DEBOUNCE_MS);
    }

    function scanAndRemoveAds() {
        try {
            removeAdElementsByClass();
            removeAdElementsByText();
            removeBannerAds();
            removeInvisibleAds();
            removeTouchTransparentAds();
        } catch (e) {
            warn('Scan error:', e);
        }
    }

    // --- 按类名移除 ---
    function removeAdElementsByClass() {
        const selectors = ['.banner-src', '.sponsor-block', '.tag-label'];
        selectors.forEach(sel => {
            document.querySelectorAll(sel).forEach(el => {
                if (!isPlayerElement(el)) {
                    safeRemove(el, 'class:' + sel);
                }
            });
        });
    }

    // --- 按文本内容移除 ---
    function removeAdElementsByText() {
        const walker = document.createTreeWalker(
            document.body,
            NodeFilter.SHOW_ELEMENT,
            null,
            false
        );
        const toRemove = [];
        let node;
        while (node = walker.nextNode()) {
            if (isPlayerElement(node)) continue;
            if (containsAdText(node)) {
                if (node.children.length < 5 && node.textContent.length < 200) {
                    toRemove.push(node);
                }
            }
        }
        toRemove.forEach(el => {
            safeRemove(el, 'text');
        });
    }

    // --- 横幅广告拦截 ---
    function removeBannerAds() {
        const vw = window.innerWidth;
        const vh = window.innerHeight;
        const allEls = document.querySelectorAll('*');

        allEls.forEach(el => {
            if (isPlayerElement(el)) return;
            const style = window.getComputedStyle(el);
            const pos = style.position;
            if (pos !== 'fixed' && pos !== 'sticky' && pos !== 'absolute') return;

            const w = el.offsetWidth;
            const h = el.offsetHeight;

            if (w > vw * 0.5 && h >= 30 && h <= 250) {
                const bottom = parseFloat(style.bottom);
                const top = parseFloat(style.top);

                if (!isNaN(bottom) && bottom <= 10) {
                    safeRemove(el, 'bottom-banner');
                    return;
                }
                if (!isNaN(top) && top <= 10) {
                    // 6.1 修复：整体属于站点导航/菜单的容器一律跳过（不再仅判断 NAV 标签）
                    if (isNavigationElement(el)) return;
                    if (el.tagName !== 'NAV' && !el.querySelector('nav') &&
                        !el.closest('nav')) {
                        safeRemove(el, 'top-banner');
                    }
                }
            }
        });
    }

    // --- 隐形广告拦截（5种类型） ---
    function removeInvisibleAds() {
        const vw = window.innerWidth;
        const vh = window.innerHeight;
        const allEls = document.querySelectorAll('*');

        allEls.forEach(el => {
            if (isPlayerElement(el)) return;
            const style = window.getComputedStyle(el);
            const opacity = parseFloat(style.opacity);
            const zIndex = parseInt(style.zIndex, 10);
            const w = el.offsetWidth;
            const h = el.offsetHeight;
            const pointerEvents = style.pointerEvents;

            // 类型A：大面积透明遮罩
            if (opacity < 0.08 && w > vw * 0.6 && h > vh * 0.6) {
                safeRemove(el, 'invisible-A');
                return;
            }

            // 类型B：高 z-index 全屏覆盖
            if (zIndex >= 999 && w > vw * 0.8 && h > vh * 0.8) {
                const hasForm = el.querySelector('form, input, textarea, select, button[type="submit"]');
                if (!hasForm) {
                    safeRemove(el, 'invisible-B');
                    return;
                }
            }

            // 类型C：隐形点击热区
            if (opacity < 0.15 && pointerEvents !== 'none' && w > 150 && h > 150) {
                safeRemove(el, 'invisible-C');
                return;
            }

            // 类型D：隐藏但占位元素
            if (style.visibility === 'hidden' &&
                (style.position === 'fixed' || style.position === 'absolute') &&
                w > vw * 0.3 && h > vh * 0.2) {
                safeRemove(el, 'invisible-D');
                return;
            }

            // 类型E：可疑中间弹窗
            if (zIndex >= 999 && w > 200 && h > 150) {
                const centerX = el.getBoundingClientRect().left + w / 2;
                const centerY = el.getBoundingClientRect().top + h / 2;
                const isCentered = Math.abs(centerX - vw / 2) < vw * 0.3 &&
                                   Math.abs(centerY - vh / 2) < vh * 0.3;
                if (isCentered) {
                    const textLen = (el.innerText || '').trim().length;
                    const hasForm = el.querySelector('input, textarea, form');
                    const hasVideo = el.querySelector('video, iframe[src*="player"]');
                    // 6.1 修复：导航/菜单容器不视为中间弹窗
                    if (!isNavigationElement(el) && textLen < 50 && !hasForm && !hasVideo) {
                        safeRemove(el, 'invisible-E');
                        return;
                    }
                }
            }
        });
    }

    // --- 触控透明广告专项检测 ---
    function removeTouchTransparentAds() {
        const vw = window.innerWidth;
        const vh = window.innerHeight;
        const allEls = document.querySelectorAll('*');

        allEls.forEach(el => {
            if (isPlayerElement(el)) return;
            const style = window.getComputedStyle(el);
            const opacity = parseFloat(style.opacity);
            const w = el.offsetWidth;
            const h = el.offsetHeight;

            // 极薄透明广告条（高度 1~30px，全宽）
            if (opacity > 0 && opacity < 0.3 && h >= 1 && h <= 30 && w > vw * 0.8) {
                const pos = style.position;
                if (pos === 'fixed' || pos === 'absolute' || pos === 'sticky') {
                    safeRemove(el, 'thin-bar');
                    return;
                }
            }

            // 全屏极低透明度覆盖层（专门针对"滑动就跳转"的透明广告）
            if (opacity > 0 && opacity < 0.05 && w > vw * 0.95 && h > vh * 0.95) {
                const hasContent = el.querySelector('video, p, h1, h2, h3, h4, h5, h6, form, table, ul, ol, li, a, img:not([src*="ad"]):not([class*="banner"])');
                if (!hasContent) {
                    safeRemove(el, 'full-invisible');
                    return;
                }
            }

            // 强制禁用可疑透明元素的 pointer-events
            if (opacity > 0 && opacity < 0.1 && w > 100 && h > 100) {
                const pos = style.position;
                if ((pos === 'fixed' || pos === 'absolute') && style.pointerEvents !== 'none') {
                    const zIndex = parseInt(style.zIndex, 10);
                    // 6.1 修复：导航容器不禁用其 pointer-events，保证菜单可点击
                    if (zIndex > 900 && !isNavigationElement(el)) {
                        log('Disabled pointer-events on suspicious element');
                        el.style.pointerEvents = 'none';
                        return;
                    }
                }
            }
        });
    }

    // ===================== CSS 注入兜底 =====================
    function injectCSS() {
        if (document.getElementById('ad-block-v6-css')) return;
        const style = document.createElement('style');
        style.id = 'ad-block-v6-css';
        style.textContent = `
            /* 已知广告类 */
            .banner-src, .sponsor-block, .tag-label,
            .ad-touch-layer, .ad-hit-area, .ad-overlay-layer,
            .picture-ad, .advertisement, .mediav_ad, .ad-box,
            .ad-banner, .ad-footer, .ad-header, .ad-float,
            .ad-popup, .ad-overlay, .ad-layer, .ad-mask,
            .advert, .advertise, .adsbygoogle,
            .gg-wrap, .gg-box, .guanggao, .tanxssp,
            .hide-article-box, .bottom_pop, #bottomPop,
            .csdn-bottom-strip-ad, .csdn-common-logo-advert,
            .pulllog-box, .login-mark, .leftPop, .rightPop,
            .ad-article-right, .side-ad, #sidebar_ad, .sidebar-ad,
            .right-ad, .aside_ad, .sidebar-recommend, .aside-ad-wrap,
            .recommend-ad-box, .ad-layer, .ad-mask,
            .stui-header__user li a[href*="javascript:"],
            .stui-header__menu li a[href*="javascript:"] {
                display: none !important;
                visibility: hidden !important;
                height: 0 !important;
                overflow: hidden !important;
                pointer-events: none !important;
            }
            /* 可疑域名图片 */

            img[src*="1727xpj"] {
                display: none !important;
            }
            /* 可疑域名链接 */
            a[href*="bcvxfsd68"], a[href*="1-727-xz"],
            a[href*="94684572994"], a[href*="wwnb8"],
            a[href*="hdflby"] {
                display: none !important;
                pointer-events: none !important;
            }
            /* 全屏固定定位透明覆盖层通用拦截 */
            div[style*="position: fixed"][style*="bottom: 0"],

    // ===================== 启动 =====================
    injectCSS();

    // 首次扫描
    scanAndRemoveAds();

    // 多时间点延迟扫描
    CONFIG.SCAN_DELAY.forEach(delay => {
        setTimeout(scanAndRemoveAds, delay);
    });

    // MutationObserver 实时监听（带防抖）
    let debounceTimer = null;
    const observer = new MutationObserver(() => {
        if (debounceTimer) return;
        debounceTimer = setTimeout(() => {
            debounceTimer = null;
            scanAndRemoveAds();
            injectCSS();
        }, CONFIG.DEBOUNCE_MS);
    });

    if (document.body) {
        observer.observe(document.body, { childList: true, subtree: true });
    } else {
        document.addEventListener('DOMContentLoaded', () => {
            observer.observe(document.body, { childList: true, subtree: true });
        });
    }

    // 窗口大小变化时重新扫描
    window.addEventListener('resize', () => {
        clearTimeout(debounceTimer);
        debounceTimer = setTimeout(scanAndRemoveAds, 300);
    });

    log('脚本已启动 (v6.0) | 当前网站:', currentHost);
})();
